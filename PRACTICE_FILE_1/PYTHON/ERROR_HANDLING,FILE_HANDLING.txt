# Question:
# Write a Python program to continuously take names from the user,
# store them in a list, and stop when the user types "stop",
# using try-except-else-finally.

s = []  # Empty list to store names

while True:
    try:
        d = input("enter your name:(you can go exit write stop)")
        if d.lower() == "stop":   # Stop condition
            break
        s.append(d)               # Add name to list
        print(s)
    except NameError:
        print("name is not defined in double quotation")
    else:
        print("welcome to the list you can add another element")
    finally:
        print("Happy to join our team")

# Comment:
# This program demonstrates:
# - Using an infinite while loop for continuous input
# - Storing user input in a list
# - Using 'stop' as a condition to exit the loop
# - try-except-else-finally for error handling
# - finally block always executes whether error occurs or not

# Question:
# Write a Python program to copy the contents of one file
# (worker.txt) into another file (worker2.txt)
# using the 'with' statement (context manager).

with open("worker.txt") as a:              # Open source file in read mode
    with open("worker2.txt", "w") as f:    # Open destination file in write mode
        for i in a:                        # Read file line by line
            f.write(i)                     # Write each line into new file
        print("File copied successfully")  # Success message

# Question:
# Write a Python program to append data to a file,
# and read its content using the 'with' statement.
# Demonstrate the use of file pointer methods: tell() and seek().

with open("C:\\Users\\DELL\\Desktop\\tops.txt", "a+") as dfg:  
    dfg.write("\n\nokk done")                # Append data to the file
    print(dfg.tell())                         # tell() shows current pointer position after writing

    dfg.seek(0)                               # Move cursor to the beginning of the file
    print("2:", dfg.tell())                    # Pointer position after seek(0)
    
    print(dfg.read())                          # Read and display the file content
    print("3:", dfg.tell())                    # Pointer position after reading
