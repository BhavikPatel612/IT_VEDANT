# Question:
# Write a Python program to combine a date object and a time object
# into a single datetime object.

from datetime import datetime, date, time

d = date(2025, 10, 10)        # Create a date object
t = time(15, 30, 14)          # Create a time object

dt = datetime.combine(d, t)   # Combine date and time into a datetime object
print(dt)                      # Display the combined datetime
