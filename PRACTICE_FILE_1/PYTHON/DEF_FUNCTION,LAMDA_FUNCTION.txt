# Question:
# Write a Python program to define a function that calculates the factorial
# of a number and prints the result.

def fact(g):
    num = 1
    fact = 1
    for h in range(num, g + 1):
        fact = fact * h
    print(fact)  # Print factorial of g

# Function call
fact(5)  # Call the function with argument 5

# Comment:
# This program demonstrates:
# - Defining a function to calculate factorial
# - Using a for loop inside a function
# - Passing an argument to the function
# - Printing the computed factorial
# - Functions allow reusing the code with different inputs

# - Functions with **kwargs are flexible for varying number of named inputs

# for multiplication and square of a number.

# Question:
# Write a Python program to sort a list of strings based on their length
# using a lambda function as the key.

names = ["Ramesh", "Bhavik", "Jayesh"]  # List of names

names.sort(key=lambda x: len(x))        # Sort by length of each name
print(names)                            # Display sorted list

# Question:
# Write a Python program to use map() with a lambda function
# to square all numbers in a list.

numbers = [1, 2, 3, 4]                          # Original list

squared = list(map(lambda x: x**2, numbers))    # Square each element using map and lambda
print(squared)                                  # Display squared numbers

# Question:
# Write a Python program to use filter() with a lambda function
# to get all even numbers from a list.

numbers = [10, 15, 20, 25, 30]               # Original list

even = list(filter(lambda x: x % 2 == 0, numbers))  # Filter even numbers using lambda
print(even)                                   # Display even numbers

# Question:
# Write a Python program to use reduce() with a lambda function
# to calculate the sum of all elements in a list.

from functools import reduce   # Import reduce function

numbers = [1, 2, 3, 4]  
sum_all = reduce(lambda x, y: x + y, numbers)  # Sum all elements using reduce and lambda
print(sum_all)  # Display the result
